\section{はじめに}
\begin{frame}{背景：CNの進化とRAN更新のギャップ}
  \begin{itemize}
    \item \textbf{コアネットワークの進化 (ソフトウェア化)}
      \begin{itemize}
        \item CNは5G以降でソフトウェア化されたため高速に進化
        \item 6Gでも5Gコアをベースに継続的な機能拡張が予想される
      \end{itemize}
    \item \textbf{現状のデプロイメントとその限界}
      \begin{itemize}
        \item 5Gコア特有の新機能(スライシング等)を活用するには、SA (Standalone) 構成へ移行する必要がある
        \item 現在はRANを先行更新するNSA (Non-Standalone)構成が主流だが、CNは4Gのままであり新機能が使えない
        \item SA化には対応RANへの更新が必要だが、コストや物理的制約(例: NTN)により困難な場合がある
      \end{itemize}
    $\Rightarrow$ \textbf{CNの進化速度に対し、RANの更新サイクルにギャップが生じる}
  \end{itemize}
\end{frame}

\begin{frame}{課題：ギャップが問題になるケース}
  \begin{itemize}
    \item \textbf{研究開発：新機能の先行検証}
      \begin{itemize}
        \item 次世代CN機能（スライシング、URLLC等）を検証したい
        \item 対応RAN（5G gNB）は高価で免許取得も必要
        \item $\Rightarrow$ \textbf{低コストかつ手軽に検証できる環境が必要}
      \end{itemize}
    \item \textbf{商用導入：マイグレーション戦略の柔軟性欠如}
      \begin{itemize}
        \item 商用網を前提にした場合、全RANを一斉に更新するのは現実的でない
        \item 段階移行のためのOptionもあるが、移行期間中は旧RANエリアで新CN機能が利用できない
        \item $\Rightarrow$ \textbf{RAN更新を伴わないCN先行移行パスが必要}
      \end{itemize}
    \item \textbf{運用維持：物理的に更新が難しいRAN}
      \begin{itemize}
        \item NTNでは衛星やHAPSを使用するため、一度配備すると地上局よりも物理的な更新が困難
        \item 地上のCNが進化しても、空のRANは旧世代のまま残る
        \item $\Rightarrow$ \textbf{新旧世代が共存可能な収容技術が必要}
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{目的とアプローチ}
  \begin{block}{目的}
    物理的に更新困難なレガシーRANを、ハードウェア変更なしに次世代CNに収容し、段階的な移行と新機能利用を可能にするアーキテクチャを提案する。
  \end{block}
  \begin{itemize}
    \item \textbf{アプローチ：プロトコル変換による疎結合化}
      \begin{itemize}
        \item 変換ゲートウェイを介在させ、RANとCNの直接的な依存を排除
        \item 世代間で不変な機能（AAA, Mobility Management, U-plane）を抽象化してマッピング
      \end{itemize}
    \item \textbf{本研究のスコープ}
      \begin{itemize}
        \item 4G eNB (LTE) と 5Gコア (5GC) の異世代相互接続を実証
      \end{itemize}
    \item \textbf{期待される効果（課題への回答）}
      \begin{itemize}
        \item \textbf{R\&D:} 既存RANを用いた安価な次世代CN検証環境
        \item \textbf{移行:} RAN更新を伴わない柔軟なCN先行移行
        \item \textbf{運用:} NTNなど物理制約のあるインフラの延命と機能拡張
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{世代不変性に基づく疎結合化の原理}
  \begin{block}{キーアイデア}
    モバイル通信の根幹機能(AAA, Mobility Management, U-plane)は世代を超えて\alert{本質的に同じ}であり、プロトコル表現が異なるだけ $\Rightarrow$ 抽象化して相互変換可能
  \end{block}
  \begin{itemize}
    \item \textbf{C-Plane (制御信号): ステートマシン変換}
      \begin{itemize}
        \item 接続管理(Attach/Registration)、Mobility Management(HO)等の状態遷移は世代不変
        \item S1-AP $\leftrightarrow$ NG-AP のメッセージ名・パラメータをマッピング
      \end{itemize}
    \item \textbf{U-Plane (データ転送): QoS抽象化}
      \begin{itemize}
        \item Bearer (4G) も QoS Flow (5G) も「優先度付きトンネル」という点で同一
        \item GTP-U TEID とQoS識別子 (EBI $\leftrightarrow$ QFI) を付け替え
      \end{itemize}
    \item \textbf{NAS (端末-CN通信): 透過転送}
      \begin{itemize}
        \item \alert{NAS over LTE:} 5G NASを4GのNASコンテナに格納して運ぶ
        \item これにより端末は4G無線上で5Gコアへ直接登録可能(技術的なブレイクスルー)
      \end{itemize}
  \end{itemize}
\end{frame}

            \item AttachReq $\rightarrow$ RegistrationReq
          \end{itemize}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{U-Plane変換}
      \begin{itemize}
        \item \textbf{QoS変換}
          \begin{itemize}
            \item QCI $\rightarrow$ 5QI
            \item E-RAB ID $\rightarrow$ PDU Session ID
          \end{itemize}
        \item \textbf{TEID管理}
          \begin{itemize}
            \item S1-U TEID $\leftrightarrow$ N3 TEID
            \item マッピングテーブル管理
          \end{itemize}
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \begin{block}{実装のポイント}
    \begin{itemize}
      \item UE毎のコンテキスト管理（ENB-UE-ID、AMF-UE-ID、TEID）
      \item NASメッセージの透過的な転送（暗号化されたまま変換）
      \item セキュリティパラメータの保持（eKSI $\leftrightarrow$ ngKSI）
    \end{itemize}
  \end{block}
\end{frame}

    \item \textbf{考察}
      \begin{itemize}

        \item $\Rightarrow$ \textbf{異世代相互接続は性能面でも実用可能}であることを確認
      \end{itemize}
  \end{itemize}

\begin{frame}{考察：デプロイメント戦略（2/2）}
  \begin{itemize}
    \item \textbf{課題1: R\&D（低コスト検証）} → \alert{コア側配置}
      \begin{itemize}
        \item 構成の単純さ最優先。PC1台でAll-in-One構成が可能
      \end{itemize}
    \item \textbf{課題2: 商用マイグレーション} → \alert{コア側配置}
      \begin{itemize}
        \item 運用コスト最優先。数千局に対し現地作業ゼロで移行可能
      \end{itemize}
    \item \textbf{課題3: NTN（物理更新困難）} → \alert{コア側配置}（基本）
      \begin{itemize}
        \item 地上局は僻地が多く、保守性を考慮しコア集約が合理的
        \item 例外：バックホール帯域逼迫時は地上局配置でスライシング早期適用
      \end{itemize}
  \end{itemize}
  \vspace{0.5em}
  $\Rightarrow$ \textbf{基本は「コア側」への集約配置が最適解}
\end{frame}

\section{結論}
\begin{frame}{本研究の成果}
  \begin{block}{達成事項}
    \begin{itemize}
      \item \textbf{疎結合アーキテクチャの確立}: RANとCNのライフサイクルを分離し、RAN側の改修なしに5GCへの接続が可能であることを実証
      \item \textbf{Early S1 Setup方式の考案}: 4G/5Gのベアラ確立タイミングの差異を吸収するシーケンスを設計・実装
      \item \textbf{実用性の実証}: sXGP実機による検証で、変換オーバーヘッドが実運用上許容範囲であることを確認
    \end{itemize}
  \end{block}
  \begin{block}{3つの課題への貢献}
    \begin{itemize}
      \item \textbf{R\&D:} sXGP + Open5GSによる低コストな5GC検証環境を実現
      \item \textbf{商用導入:} eNBの改修なしに5GCへ接続可能な「CN先行移行」戦略を提示
      \item \textbf{NTN:} 地上ゲートウェイのみで衛星/HAPS上RANの5GC収容が可能
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{今後の展望と未実装機能}
  \begin{itemize}

      \end{itemize}

\end{frame}
