  \begin{block}{問題：スケーラビリティ}
    \begin{itemize}
      \item 1つのCN（集約配置）に対して、数百～数万のRAN（分散配置）が接続
      \item \alert{RAN側の改修では、改修コストが累積してスケールしない}
      \item 標準仕様は両者の世代同期を前提 → eNB無改修のまま5GCへ収容する標準手段がない
    \end{itemize}
  \end{block}

  \begin{block}{研究の本質：論理的基盤の構築}

      \item 4G/5G間で\textbf{不変な本質機能}（AAA・移動管理・U-Plane）を6つの観点から多角分析
      \item 5GC固有機能を\textbf{3層モデル}で階層的に整理し、各層の対応方針を導出
      \item プロトコル変換によるCN先行移行が\textbf{原理的に可能}であることを理論的に示す
    \end{itemize}
  \end{block}

  \begin{block}{実装・検証：理論の有効性を実証}
    C言語プロトタイプを実装し、実機で接続成功を確認。\\
    遅延増加3.4 ms、通信速度ベースライン同等 → 理論が実現可能であることを実証
  \end{block}

  \begin{block}{意義}
    世代共通機能に基づいたスケーラブルな移行モデル。\\
    将来の6G対応や衛星RAN統合に応用可能な汎用フレームワーク。
  \end{block}
\end{frame}

    \begin{itemize}
      \item \textbf{4G（一括）}: 登録要求 → 認証 → 暗号化 → コンテキスト設定（+登録承認+ベアラ情報）→ U-Plane確立
      \item \textbf{5G（分離）}: 登録要求 → 認証 → 暗号化 → 登録承認 → 登録完了 → \alert{その後に}セッション確立
    \end{itemize}
    $\Rightarrow$ 4Gでは登録とベアラ確立が一括、5Gでは\alert{明確に分離}
  \end{block}
\end{frame}

\begin{frame}{標準技術の制約}
  \begin{block}{N26 (CN間連携)}
    \small
    \begin{itemize}
      \item EPC と 5GC の\textbf{並存が必須} → CN先行移行不可
      \item 4G RAN配下では5GC機能が利用できない → スライシング等は5Gエリアのみ
    \end{itemize}
  \end{block}

  \begin{block}{ng-eNB (RAN更新)}
    \small
    \begin{itemize}
      \item 基地局\textbf{ファームウェア/ハードウェア改修が必須}
      \item RAN は数百～数万の基地局で構成 → 改修対象が多い
      \item \alert{改修完了まで CN を5G化できない} → スケーラビリティ問題
    \end{itemize}
  \end{block}

  \begin{block}{N3IWF (非3GPPアクセス)}
    \small
    \begin{itemize}
      \item S1インターフェース（S1AP/GTP-U）\textbf{非対応} → 既存 eNB 接続不可
      \item UE 側に\textbf{IPsecクライアント機能要求} → 既存4G UE利用不可
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{学術研究の制約と本研究の差別化}
  \begin{block}{5G-Flow (SDNベース)}
    \small
    RAN・UE\textbf{双方で OpenFlowスイッチとプロトコルスタック分割が必須}
    → 既存機器の改修が必要
  \end{block}

  \begin{block}{Hybrid 5G Core (標準拡張)}
    \small
    MME・SGW を\textbf{最小構成で残存させる必要} → EPC完全廃止は不可

    本研究との差異: 提案手法は\textbf{EPC完全廃止}を可能に
  \end{block}

  \begin{block}{提案手法の独自性}
    \small
    \begin{itemize}
      \item RAN・UE完全無改修で、\textbf{CN側のみ}に変換ゲートウェイを配置
      \item \textbf{スケーラブル}: 1台の変換GW で数百～数万の eNB に対応
      \item \textbf{世代共通機能の体系化}により、4G/5Gプロトコル間の具体的対応を実装
    \end{itemize}
  \end{block}
\end{frame}

    \centering
    \small

    $\Rightarrow$ プロトコル表現が異なるだけ \\
    $\Rightarrow$ 抽象化により相互変換が可能
  \end{alertblock}

\begin{frame}{理論的基盤：世代共通機能の多角的分析}
  \begin{block}{アプローチ}
    本質的機能の対応を\textbf{6つの観点}から体系的に分析
  \end{block}

  \begin{table}
    \small
    \begin{tabular}{|l|l|l|l|}
      \hline
      \textbf{観点} & \textbf{4G (EPS)} & \textbf{5G (5GS)} & \textbf{対応} \\
      \hline
      \textbf{機能} & AAA, 移動管理, U-Plane & \textbf{同左} & \alert{完全一致} \\
      \hline
      \textbf{ノード} & HSS, MME, S/P-GW & AUSF/UDM, AMF/SMF, UPF & 1対多分解 \\
      \hline
      \textbf{プロトコル} & IMSI, Attach, GTP-U & SUPI, Registration, GTP-U & 意味的等価 \\
      \hline
      \textbf{状態} & EMM/ECM状態 & 5GMM/CM状態 & 1対1対応 \\
      \hline
      \textbf{IF境界} & S1-MME, S1-U & N2, N3 & 機能的等価 \\
      \hline
    \end{tabular}
  \end{table}

  \vspace{0.2cm}
  $\Rightarrow$ 6つの観点すべてで明確な対応関係 \\
  $\Rightarrow$ \textbf{プロトコル変換が原理的に可能}
\end{frame}

\begin{frame}{5GC固有機能の階層分析（3層モデル）}
  \begin{table}
    \small
    \begin{tabular}{|l|l|l|}
      \hline
      \textbf{層} & \textbf{機能例} & \textbf{対応方針} \\
      \hline
      基盤層 & SBA, CUPS, NFV & \textbf{隠蔽}（CN内部事情） \\
      \hline
      制御メカニズム層 & スライス選択, QoS Flow & \textbf{変換・整合}（主たる貢献） \\
      \hline
      応用サービス層 & Network Slicing, 5G LAN & \textbf{段階的提供} \\
      \hline
    \end{tabular}
  \end{table}

  \vspace{0.3cm}
  \begin{block}{対応の階層戦略}
    \begin{itemize}
      \item \textbf{基盤層}: RAN側から不可視 $\Rightarrow$ 対応不要
      \item \textbf{制御メカニズム層}: QCI$\leftrightarrow$5QI、APN$\rightarrow$S-NSSAIマッピング $\Rightarrow$ \alert{本研究の核心}
      \item \textbf{応用サービス層}: メカニズム層の解決により享受可能
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{世代不変性に基づく疎結合化の原理}
  \begin{block}{キーアイデア}
    モバイル通信の根幹機能(AAA, Mobility Management, U-plane)は世代を超えて\alert{本質的に同じ}であり、プロトコル表現が異なるだけ $\Rightarrow$ 抽象化して相互変換可能
  \end{block}
  \begin{itemize}
    \item \textbf{C-Plane (制御信号): ステートマシン変換}
      \begin{itemize}
        \item 接続管理(Attach/Registration)、Mobility Management(HO)等の状態遷移は世代不変
        \item S1-AP $\leftrightarrow$ NG-AP のメッセージ名・パラメータをマッピング
      \end{itemize}
    \item \textbf{U-Plane (データ転送): QoS抽象化}
      \begin{itemize}
        \item Bearer (4G) も QoS Flow (5G) も「優先度付きトンネル」という点で同一
        \item GTP-U TEID とQoS識別子 (QCI $\leftrightarrow$ 5QI) を変換
      \end{itemize}
  \end{itemize}
\end{frame}

  \begin{itemize}
    \item \textbf{課題1: R\&D（低コスト検証）} → \alert{コア側配置}
      \begin{itemize}
        \item 構成の単純さ最優先。PC1台でAll-in-One構成が可能
      \end{itemize}
    \item \textbf{課題2: 商用マイグレーション} → \alert{コア側配置}
      \begin{itemize}
        \item 運用コスト最優先。数千局に対し現地作業ゼロで移行可能
      \end{itemize}
    \item \textbf{課題3: NTN（物理更新困難）} → \alert{コア側配置}（基本）
      \begin{itemize}
        \item 地上局は僻地が多く、保守性を考慮しコア集約が合理的
        \item 例外：バックホール帯域逼迫時は地上局配置でスライシング早期適用
      \end{itemize}
  \end{itemize}
  \vspace{0.5em}
  $\Rightarrow$ \textbf{基本は「コア側」への集約配置が最適解}
\end{frame}

\section{結論}
\begin{frame}{本研究の成果：理論的基盤と実証（その1）}
  \begin{block}{\textbf{G1}: 世代不変性に基づくプロトコルマッピングの体系化}
    4G/5Gプロトコル（S1AP↔NGAP、NAS EPS↔NAS 5GS、GTP-U）を比較分析。\\
    \textbf{世代共通機能}（AAA・移動管理・U-Plane）に基づく変換規則を体系化：
    \begin{itemize}
      \small
      \item NAS変換規則、ID変換規則、TEIDマッピング、QoSマッピングを設計
      \item 実機検証で信号レベルでの成立を確認 → 理論の有効性を実証
    \end{itemize}
  \end{block}

  \begin{block}{\textbf{G2}: 疎結合アーキテクチャの設計と実証}
    S1/NGインターフェース相互変換ゲートウェイを設計・実装。\\
    Early S1 Setup方式により、4G/5Gのシーケンス差異を吸収：
    \begin{itemize}
      \small
      \item 実機環境（sXGP + Open5GS + 商用UE）で検証
      \item 遅延増加3.4 ms、スループット同等 → 実運用上の実用性を確認
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{本研究の成果：拡張性と応用（その2）}
  \begin{block}{\textbf{G3}: 5GC固有機能への拡張性の明確化}
    5GC固有機能を\textbf{3層モデル}で階層的に分類し、層ごとの対応方針を設計：
    \begin{itemize}
      \small
      \item 基盤層（透過）、制御メカニズム層（マッピング）、応用サービス層（スタブ）
      \item スライシング等への拡張が既存設計の延長線上で可能であることを示す
    \end{itemize}
  \end{block}

  \begin{block}{研究の位置づけと応用}
    \begin{itemize}
      \small
      \item 低コスト5GC検証環境（sXGP + Open5GS）
      \item CN先行移行戦略（eNB無改修）
      \item NTN適用の可能性（衛星・HAPS RAN の5G化）
    \end{itemize}
  \end{block}
\end{frame}
